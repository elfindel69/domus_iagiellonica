on_character_death = {
	on_actions = { doi_on_french_monarch_death }
}


doi_on_french_monarch_death = {
	effect = {
		if = {
			limit = {
				is_monarch = yes
				exists = c:FRA
				owner = c:FRA
				owner = {
					has_law = law_type:law_monarchy
					NOT = { has_variable = kings_death }
				}
			}
			owner = {
				trigger_event = doi_france.1
			}
		
		} 
	}
}
